Structure: Bitrix24 robot → WordPress REST (PHP) → WooCommerce order → .NET 8 service (Subiekt nexo SDK) over a WireGuard tunnel → ERP document → write-back to the CRM timeline.

php — webhook: create order, forward to the ERP service

register_rest_route('fedrs/v1', '/b24/order', [
  'methods'  => 'POST',
  'callback' => function (WP_REST_Request $r) {
    $deal  = $r->get_json_params();                 // { items[], currency }
    $order = wc_create_order(['status' => 'processing']);
    foreach ($deal['items'] as $i)
      $order->add_product(wc_get_product($i['sku']), $i['qty']);
    $order->save();
    wp_remote_post(getenv('ERP_URL') . '/invoice', [  // .NET service over WireGuard
      'headers' => ['X-Auth' => getenv('ERP_SECRET')],
      'body'    => wp_json_encode(['order' => $order->get_id()]),
    ]);
    return ['ok' => true, 'order' => $order->get_id()];
  },
  'permission_callback' => 'fedrs_verify_secret',
]);

python — alert on stale WireGuard handshakes (cron, every 3 min)

import subprocess, time, requests, os

raw = subprocess.check_output(['wg', 'show', 'all', 'latest-handshakes']).decode()
for line in raw.splitlines():

python — advance a CAPA stage, log audit, find similar cases (pgvector)

@router.callback_query(F.data.startswith('capa:'))
async def capa_step(cb: CallbackQuery):
    incident_id = int(cb.data.split(':')[1])
    await db.execute('UPDATE incidents SET stage = stage + 1 WHERE id = $1', incident_id)
    await db.execute('INSERT INTO audit(incident, user_id, ts) VALUES ($1, $2, now())',
                     incident_id, cb.from_user.id)                 # immutable journal
    similar = await db.fetch(
        'SELECT id, title FROM incidents ORDER BY embedding <=> $1 LIMIT 5',
        await embed(cb.message.text))                             # cosine distance
    await cb.answer()

php — Laravel: store a submission and mirror it to Google Sheets

public function store(TrainingRequest $req)
{
    $sub = Submission::create($req->validated() + ['score' => $this->grade($req)]);

    Sheets::spreadsheet(config('sheets.id'))->sheet('Couriers')
          ->append([[ $sub->name, $sub->phone, $sub->score, now() ]]);

    return response()->json(['id' => $sub->id], 201);
}
